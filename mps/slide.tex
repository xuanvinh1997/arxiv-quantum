% WHAT IS MPS
% ══════════════════════════════════════
\begin{frame}{Matrix Product States là gì?}

Trạng thái lượng tử $N$ hạt đòi hỏi $d^N$ số phức để mô tả đầy đủ — \warn{bùng nổ tổ hợp}. MPS biểu diễn hiệu quả qua tích chuỗi ma trận:

\eqn{
  \ket{\psi} = \sum_{s_1,\ldots,s_N} A^{s_1}_1 A^{s_2}_2 \cdots A^{s_N}_N \;\ket{s_1 s_2 \cdots s_N}
}

\vspace{2pt}
\begin{center}
\begin{tikzpicture}[scale=0.9, transform shape,
  tensor/.style={draw=qcyan, fill=qcyan!15, rounded corners=3pt, minimum width=0.9cm, minimum height=0.9cm, font=\small\bfseries, text=qcyan},
  leg/.style={thick, color=dimtext},
  phys/.style={thick, color=qpurple}
]
  \foreach \i/\s in {1/$A_1$, 2/$A_2$, 3/$A_3$, 4/$\cdots$, 5/$A_N$} {

    \draw[phys] (A\i.south) -- ++(0,-0.6) node[below, font=\footnotesize, text=qpurple] {$s_{\i}$};
  }
  \draw[leg] (A1.east) -- (A2.west) node[midway, above, font=\tiny, text=dimtext] {$D$};
  \draw[leg] (A2.east) -- (A3.west) node[midway, above, font=\tiny, text=dimtext] {$D$};
  \draw[leg] (A3.east) -- (A4.west);
  \draw[leg] (A4.east) -- (A5.west) node[midway, above, font=\tiny, text=dimtext] {$D$};
\end{tikzpicture}
\end{center}

\vspace{2pt}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
  \begin{itemize}
    \item $A^{s_i}_i \in \mathbb{C}^{D\times D}$: ma trận với \highlight{bond dimension} $D$
    \item Tham số: $O(N D^2 d)$ thay vì $O(d^N)$
  \end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
  \begin{itemize}
    \item Chỉ số vật lý $s_i \in \{0,\ldots,d{-}1\}$, chỉ số ảo nối các tensor
    \item Biên mở: $A_1 \in \mathbb{C}^{1\times D}$, $A_N \in \mathbb{C}^{D\times 1}$
  \end{itemize}
\end{column}
\end{columns}

\vspace{4pt}
{\small\textcolor{dimtext}{$D=1$: product state. $D$ tăng $\to$ entanglement tăng. Entanglement entropy bị chặn: $S \leq \log D$.}}

\end{frame}

% ══════════════════════════════════════
% MPS PSEUDOCODE
% ══════════════════════════════════════
\begin{frame}[fragile, shrink=5]{Xây dựng MPS từ state vector — SVD liên tiếp}

Ý tưởng: reshape + SVD tại từng vị trí, cắt bớt bond dimension xuống $D_{\max}$.

\vspace{4pt}
\begin{lstlisting}
import numpy as np

def state_to_mps(psi, N, d, D_max=32):
    """
    psi      : state vector, shape (d**N,)
    N, d     : so site, kich thuoc khong gian cuc bo
    D_max    : bond dimension toi da (truncation)
    Tra ve   : list N tensor A[i], shape (D_left, d, D_right)
    """
    tensors = []
    remainder = psi.reshape(1, d**N)  # (D_left=1, d^N)
    D_left = 1

    for i in range(N - 1):
        # Gop chi so trai + vat ly, de SVD
        remainder = remainder.reshape(D_left * d, d**(N - 1 - i))

        U, S, Vh = np.linalg.svd(remainder, full_matrices=False)

        D = min(len(S), D_max)         # cat bot singular values nho
        U, S, Vh = U[:, :D], S[:D], Vh[:D, :]

        A = U.reshape(D_left, d, D)    # tensor site i: (D_left, d, D)
        tensors.append(A)

        remainder = np.diag(S) @ Vh    # phan con lai, chuyen sang site ke
        D_left = D

    # Site cuoi: khong can SVD nua
    tensors.append(remainder.reshape(D_left, d, 1))
    return tensors  # [A_1, A_2, ..., A_N]
\end{lstlisting}

\vspace{2pt}
{\small\textcolor{dimtext}{%
\textbf{Độ phức tạp:} $O(N d^N D)$ lần đầu — nhưng mục đích là để $D \ll d^{N/2}$, tránh bùng nổ.
\quad\textbf{Left-canonical:} $\sum_s A^{s\dagger}A^s = \mathbb{I}$ tự động từ cột của $U$.}}

\end{frame}

\begin{frame}{Giới hạn của MPS}
\begin{columns}[T]
\begin{column}{0.48\textwidth}

  \textcolor{qpink}{\textbf{1. Entanglement bị chặn}}\\[2pt]
  {\small $S \leq \log D$ — MPS chỉ biểu diễn trạng thái có vướng víu thấp.\\[3pt]
  Vi phạm định luật diện tích $\Rightarrow$ $D$ tăng theo hàm mũ:}
  \vspace{-2pt}
  \begin{itemize}\small
    \item Critical 1D: $S\sim\tfrac{c}{3}\log N \Rightarrow D\sim N^{c/6}$
    \item 2D bất kỳ: $S\sim L \Rightarrow D\sim e^{L}$
  \end{itemize}

  \vspace{6pt}
  \textcolor{qpink}{\textbf{2. Bức tường vướng víu trong diễn tiến thời gian}}\\[2pt]
  {\small Sau sự thay đổi đột ngột của tham số Hamiltonian: $S(t)\sim\alpha t$, nên $D(t)\sim e^{\alpha t}$.\\
  MPS \warn{thất bại} sau thời gian ngắn $t^* \sim \frac{\log D}{\alpha}$.}

  \vspace{6pt}
  \textcolor{qpink}{\textbf{3. Thuần túy 1D}}\\[2pt]
  {\small Cấu trúc chuỗi tuyến tính. Hệ ladder, cylinder, hay 2D đòi bond dimension tăng mũ hoặc phải chuyển sang PEPS.}

\end{column}
\begin{column}{0.48\textwidth}

  \textcolor{qpink}{\textbf{4. Co rút hiệu quả — chỉ may mắn ở 1D}}\\[2pt]
  {\small MPS: $O(ND^3)$ — khả thi.\\
  Mở rộng sang PEPS 2D: \warn{\#P-khó} — mọi tính toán đều phải xấp xỉ.}

  \vspace{6pt}
  \textcolor{qpink}{\textbf{5. Trạng thái kích thích \& hỗn hợp}}\\[2pt]
  {\small Tối ưu tốt nhất cho trạng thái cơ bản.\\
  Trạng thái kích thích cần bổ sung điều kiện trực giao — chi phí tăng theo số trạng thái.\\
  Nhiệt độ hữu hạn: cần toán tử tích ma trận (MPO), tốn $D^2$ lần hơn.}

  \vspace{6pt}
  \textcolor{qpink}{\textbf{6. Dấu Fermion \& hệ thất vọng}}\\[2pt]
  {\small Cần chuỗi Jordan--Wigner cho Fermion — tính cục bộ bị phá vỡ trong 2D.\\
  Mạng tinh thể thất vọng: vấn đề dấu vẫn có thể xuất hiện.}

  \vspace{6pt}
  {\small\textcolor{dimtext}{$\Rightarrow$ Mỗi giới hạn mở ra một hướng mở rộng:\\MERA, PEPS, NQS, cMPS, isoTNS\ldots}}

\end{column}
\end{columns}
\end{frame}

\begin{frame}{MERA — Vướng víu logarithmic}

{\small Hệ tới hạn: $S(L)\sim\frac{c}{3}\log L$ — MPS cần $D\to\infty$.
Vidal (2007): xen \highlight{disentangler} $U$ trước mỗi bước thô hóa isometry $W$:}
\vspace{-2pt}
\eqn{\ket{\psi_{\text{MERA}}} = \Bigl(\prod_{\tau} U_\tau W_\tau\Bigr)\ket{0}^{\otimes N}}
\vspace{-4pt}

\begin{columns}[T]
\begin{column}{0.34\textwidth}
  \textcolor{qcyan}{\textbf{Cải tiến cốt lõi}}
  \begin{itemize}\small
    \item $U$ loại bỏ vướng víu tầm ngắn trước khi coarse-grain
    \item Nón nhân quả $O(\log N)$ tensor — tính quan sát cục bộ hiệu quả
    \item Bắt được entropy logarithmic với $D$ hữu hạn
  \end{itemize}
\end{column}
\begin{column}{0.32\textwidth}
  \textcolor{qgreen}{\textbf{Hiệu quả}}
  \begin{itemize}\small
    \item Trích xuất $\Delta$, $c$ của CFT từ siêu toán tử chuyển
    \item MERA $\leftrightarrow$ AdS/CFT: hình học hyperbolic tự nhiên
    \item Thực nghiệm: Trotterized MERA trên Quantinuum H1 (2023)
  \end{itemize}
\end{column}
\begin{column}{0.30\textwidth}
  \textcolor{qpink}{\textbf{Hạn chế}}
  \begin{itemize}\small
    \item Chi phí $O(\chi^8)$ mỗi vòng quét — đắt hơn DMRG nhiều lần
    \item Khó mở rộng sang 2D
    \item Tối ưu hóa không ổn định với $\chi$ lớn
  \end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{PEPS — Tensor network 2D}

{\small Mỗi tensor rank-5: $T^{s}_{l,r,u,d}$. Biểu diễn được \highlight{định luật diện tích 2D}: $S \leq |\partial A|\log D$.}

\begin{columns}[T]
\begin{column}{0.34\textwidth}
  \textcolor{qcyan}{\textbf{Cải tiến cốt lõi}}
  \begin{itemize}\small
    \item Mở rộng cấu trúc MPS sang mạng 2D tự nhiên
    \item Biểu diễn chính xác trật tự tôpô: toric code, string-net qua $G$-injective PEPS
    \item Fermion 2D qua fPEPS (Corboz)
  \end{itemize}
\end{column}
\begin{column}{0.32\textwidth}
  \textcolor{qgreen}{\textbf{Hiệu quả}}
  \begin{itemize}\small
    \item Co rút xấp xỉ boundary MPS: $O(\chi_{\text{env}}^3 D^4)$
    \item AD-based iPEPS (2024): tự vi phân, tối ưu hóa tự động
    \item Mô phỏng Hubbard 2D vượt QMC về dấu
  \end{itemize}
\end{column}
\begin{column}{0.30\textwidth}
  \textcolor{qpink}{\textbf{Hạn chế}}
  \begin{itemize}\small
    \item Co rút chính xác: \warn{\#P-khó} — mọi tính toán đều xấp xỉ
    \item Lỗi co rút khó kiểm soát, khó ước lượng sai số
    \item Tối ưu hóa kém ổn định hơn MPS
  \end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Hybrid MPS + NQS}

{\small MPS: cấu trúc rõ, nhưng bị chặn entanglement. NQS: linh hoạt, nhưng thiếu cấu trúc. Kết hợp:}
\vspace{-2pt}
\eqn{\psi(\mathbf{s}) = \underbrace{\bra{L}A^{s_1}\cdots A^{s_N}\ket{R}}_{\text{khung xương MPS}}\;\cdot\;\underbrace{f_\theta(\mathbf{s})}_{\text{hiệu chỉnh neural}}}
\vspace{-4pt}

\begin{columns}[T]
\begin{column}{0.34\textwidth}
  \textcolor{qcyan}{\textbf{Cải tiến cốt lõi}}
  \begin{itemize}\small
    \item Mạng neural hiệu chỉnh phần vướng víu mà MPS bỏ sót
    \item Phân cấp chặt: MPS $\subsetneq$ PEPS $\subsetneq$ Deep NQS
    \item Born machine: sampling chính xác $O(ND^2d)$
  \end{itemize}
\end{column}
\begin{column}{0.32\textwidth}
  \textcolor{qgreen}{\textbf{Hiệu quả}}
  \begin{itemize}\small
    \item ANTN (NeurIPS 2023): sai số $<10^{-4}$ cho hệ frustrated
    \item Neuralized fermionic TN (2026): vượt fPEPS trên Hubbard 2D
    \item Sampling tự hồi quy không cần MCMC
  \end{itemize}
\end{column}
\begin{column}{0.30\textwidth}
  \textcolor{qpink}{\textbf{Hạn chế}}
  \begin{itemize}\small
    \item Mất tính diễn giải của cấu trúc tensor
    \item Huấn luyện không ổn định, dễ gặp cực tiểu địa phương
    \item Khó kiểm soát và định lượng sai số
  \end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Isometric TN — Ràng buộc mở ra hai kết quả lớn}

{\small PEPS + ràng buộc isometric $\sum_s A^{s\dagger}A^s=\mathbb{I}$ theo một hướng $\Rightarrow$ co rút hiệu quả (Zaletel--Pollmann 2020).}

\begin{columns}[T]
\begin{column}{0.34\textwidth}
  \textcolor{qcyan}{\textbf{Cải tiến cốt lõi}}
  \begin{itemize}\small
    \item Ràng buộc isometric giữ nguyên norm khi co rút một chiều
    \item Mở rộng được sang 2D mà vẫn tính được quan sát
    \item Biểu diễn chính xác string-net liquid
  \end{itemize}
\end{column}
\begin{column}{0.32\textwidth}
  \textcolor{qgreen}{\textbf{Hiệu quả}}
  \begin{itemize}\small
    \item Co rút khả thi — không cần xấp xỉ như PEPS
    \item \good{Không có barren plateau} (Barthel--Miao 2024): $\operatorname{Var}(\partial_\theta E)\geq\Omega\!\left(\frac{1}{\mathrm{poly}(N)}\right)$
    \item Áp dụng cho mọi isoTN: MPS, TTNS, MERA
  \end{itemize}
\end{column}
\begin{column}{0.30\textwidth}
  \textcolor{qpink}{\textbf{Hạn chế}}
  \begin{itemize}\small
    \item Tính quan sát isoTNS: \warn{BQP-complete} (Malz--Trivedi 2025)
    \item Khả năng biểu diễn kém hơn PEPS đầy đủ
    \item Ràng buộc isometric khó duy trì trong tối ưu hóa
  \end{itemize}
\end{column}
\end{columns}
\end{frame}

\vspace{4pt}
\begin{center}\small

  \end{itemize}

$f(\mathbf{x}) = \bra{\psi_W}\bigotimes_i\ket{\Phi(x_i)}$, với $\ket{\Phi(x)}=\cos\tfrac{\pi x}{2}\ket{0}+\sin\tfrac{\pi x}{2}\ket{1}$.}

\begin{columns}[T]

\begin{column}{0.32\textwidth}

\end{column}
\begin{column}{0.30\textwidth}

  \begin{itemize}\small
    \item Cấu trúc 1D không tự nhiên với dữ liệu thực
    \item Kém linh hoạt hơn mạng neural sâu trên dữ liệu phức tạp

% ══════════════════════════════════════
% 7. TIME-DEPENDENT
% ══════════════════════════════════════

\begin{frame}{Diễn tiến thời gian — TEBD vs TDVP}

\begin{columns}[T]
\begin{column}{0.34\textwidth}
  \textcolor{qcyan}{\textbf{Cải tiến cốt lõi}}\\[2pt]
  {\small \textbf{TEBD} (Vidal 2003): Trotter phân rã $e^{-iH\delta t}$, mỗi cổng áp lên bond rồi cắt bớt SVD.}\\[4pt]
  {\small \textbf{TDVP} (Haegeman 2011): chiếu Schrödinger lên không gian tiếp tuyến $\mathcal{M}_D$:}
  \vspace{-2pt}
  \eqn{i\partial_t\ket{\psi}=\hat{P}_{T_{\ket{\psi}}\mathcal{M}}H\ket{\psi}}
  \vspace{-6pt}
  {\small 1TDVP bảo toàn năng lượng; 2TDVP cho $D$ thích ứng.}
\end{column}
\begin{column}{0.32\textwidth}
  \textcolor{qgreen}{\textbf{Hiệu quả}}
  \begin{itemize}\small
    \item TDVP: không lỗi Trotter, bảo toàn các đại lượng bảo toàn
    \item TEBD: đơn giản, song song hóa tốt
    \item Chi phí $O(ND^3d^3)$ — cạnh tranh cho hệ 1D
  \end{itemize}
\end{column}
\begin{column}{0.30\textwidth}
  \textcolor{qpink}{\textbf{Hạn chế cơ bản}}
  \begin{itemize}\small
    \item Sau sự thay đổi đột ngột của tham số Hamiltonian: $S(t)\sim\alpha t\Rightarrow D(t)\sim e^{\alpha t}$
    \item MPS \warn{thất bại} sau $t^*\sim\frac{\log D}{\alpha}$
    \item Hướng mở: influence functional, temporal entanglement
  \end{itemize}
\end{column}
\end{columns}
\end{frame}

% ══════════════════════════════════════
% 8. iMPS
% ══════════════════════════════════════

\begin{frame}{MPS vô hạn đồng nhất (iMPS)}

{\small MPS đồng nhất: $\cdots A^{s_{-1}}A^{s_0}A^{s_1}\cdots$ — cùng một tensor $A$ lặp lại. Không cần giới hạn hệ hữu hạn.}

\begin{columns}[T]
\begin{column}{0.34\textwidth}
  \textcolor{qcyan}{\textbf{Cải tiến cốt lõi}}
  \begin{itemize}\small
    \item Nhiệt động lực học chính xác không có sai số finite-size
    \item Transfer matrix $\mathbb{T}=\sum_s A^s\otimes\bar{A}^s$: độ dài tương quan $\xi=-1/\ln|\lambda_2|$
    \item Quasiparticle ansatz cho kích thích: $\ket{\Phi_p(B)}=\sum_n e^{ipn}\cdots B^{s_n}\cdots$
  \end{itemize}
\end{column}
\begin{column}{0.32\textwidth}
  \textcolor{qgreen}{\textbf{Hiệu quả}}
  \begin{itemize}\small
    \item VUMPS (2018): hội tụ nhanh hơn iDMRG đáng kể
    \item Tính dispersion $E(p)$ và ma trận S từ ansatz quasiparticle
    \item Fixed-point: $A_C = CA_R = A_LC$ — canonical form rõ ràng
  \end{itemize}
\end{column}
\begin{column}{0.30\textwidth}
  \textcolor{qpink}{\textbf{Hạn chế}}
  \begin{itemize}\small
    \item Chỉ áp dụng cho hệ đồng nhất (translation invariant)
    \item Chuyển pha: cần cẩn thận về vỡ đối xứng tự phát
    \item Không trực tiếp mô phỏng biên và tạp chất
  \end{itemize}
\end{column}
\end{columns}
\end{frame}

% ══════════════════════════════════════
% 9. FERMIONIC & SYMMETRIC
% ══════════════════════════════════════

\begin{frame}{MPS Fermionic \& Đối xứng}

{\small Khai thác đối xứng trong cấu trúc tensor để giảm chi phí tính toán theo hàm mũ.}

\begin{columns}[T]
\begin{column}{0.34\textwidth}
  \textcolor{qcyan}{\textbf{Cải tiến cốt lõi}}
  \begin{itemize}\small
    \item Fermion: đại số $\mathbb{Z}_2$-graded, $a\otimes b\to(-1)^{|a||b|}b\otimes a$ — không cần chuỗi Jordan--Wigner tường minh
    \item U(1): $[Q,H]=0\Rightarrow$ tensor block-diagonal
    \item SU(2): Wigner--Eckart — chỉ lưu phần tử rút gọn
  \end{itemize}
\end{column}
\begin{column}{0.32\textwidth}
  \textcolor{qgreen}{\textbf{Hiệu quả}}
  \begin{itemize}\small
    \item U(1): tăng tốc $\sim Q^2$ lần
    \item SU(2): thêm $\bar{d}^3$ lần — tổng hợp rất lớn
    \item Đọc trực tiếp tham số trật tự SPT từ biểu diễn chiếu trên chỉ số ảo
  \end{itemize}
\end{column}
\begin{column}{0.30\textwidth}
  \textcolor{qpink}{\textbf{Hạn chế}}
  \begin{itemize}\small
    \item Chuỗi Jordan--Wigner vẫn phá tính cục bộ trong 2D
    \item Non-Abelian phức tạp khi lập trình, dễ sai
    \item Đối xứng không khả nghịch (2026) vẫn là hướng mở
  \end{itemize}
\end{column}
\end{columns}
\end{frame}

% ══════════════════════════════════════
% 10. cMPS
% ══════════════════════════════════════

\begin{frame}{cMPS — MPS trong giới hạn liên tục}

{\small Giới hạn liên tục của MPS: thay tensor rời rạc bằng ma trận $Q,R_\alpha\in M_D(\mathbb{C})$ phụ thuộc vị trí:}
\vspace{-4pt}
\eqn{
  \ket{\psi_{\text{cMPS}}} = \operatorname{Tr}_{\text{aux}}\!\left[\mathcal{P}\exp\!\int_0^L\!\mathrm{d}x\,\Bigl(Q\otimes\mathbb{I}+\sum_\alpha R_\alpha\otimes\hat{\psi}^\dagger_\alpha(x)\Bigr)\right]\ket{\Omega}
}
\vspace{-6pt}

\begin{columns}[T]
\begin{column}{0.34\textwidth}
  \textcolor{qcyan}{\textbf{Cải tiến cốt lõi}}
  \begin{itemize}\small
    \item Biểu diễn trực tiếp trạng thái trường lượng tử — không cần mạng tinh thể
    \item Bất biến gauge: $Q\to GQG^{-1}+(\partial G)G^{-1}$
    \item RCMPS (Tilloy 2021): thỏa mãn mọi tiên đề Feynman, không cần cutoff UV
  \end{itemize}
\end{column}
\begin{column}{0.32\textwidth}
  \textcolor{qgreen}{\textbf{Hiệu quả}}
  \begin{itemize}\small
    \item Lieb--Liniger: độ chính xác $\sim10^{-6}$ với $D\approx64$
    \item cMERA: nhóm tái chuẩn hóa vướng víu liên tục $\leftrightarrow$ holographic RG
    \item Tính toán non-perturbative cho QFT 1+1D
  \end{itemize}
\end{column}
\begin{column}{0.30\textwidth}
  \textcolor{qpink}{\textbf{Hạn chế}}
  \begin{itemize}\small
    \item cPEPS cho QFT 2+1D vẫn là bài toán mở
    \item Tối ưu hóa phức tạp hơn MPS rời rạc
    \item Chưa có ứng dụng thực tiễn ngoài 1+1D
  \end{itemize}
\end{column}
\end{columns}
\end{frame}
